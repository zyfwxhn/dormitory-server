-- ----------------------------
-- 1. 学生表
-- ----------------------------
CREATE TABLE `student` (
  `id` bigint NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
  `student_no` varchar(32) NOT NULL COMMENT '学号（唯一索引，用于登录）',
  `password` varchar(128) NOT NULL COMMENT '密码（MD5加密存储）',
  `name` varchar(32) NOT NULL COMMENT '姓名',
  `gender` char(1) DEFAULT NULL COMMENT '性别（0:女, 1:男）',
  `grade` varchar(20) DEFAULT NULL COMMENT '年级（如：大一、研二等）',
  `phone` varchar(11) DEFAULT NULL COMMENT '手机号',
  `building_no` varchar(32) DEFAULT NULL COMMENT '楼栋号',
  `room_no` varchar(32) DEFAULT NULL COMMENT '宿舍号',
  `avatar` varchar(255) DEFAULT NULL COMMENT '头像链接 (OSS地址)',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '记录创建时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '记录最后修改时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `idx_student_no` (`student_no`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='学生信息表';

-- ----------------------------
-- 2. 报修订单表
-- ----------------------------
CREATE TABLE `repair_order` (
  `id` bigint NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
  `student_id` bigint NOT NULL COMMENT '报修学生ID',
  `repair_type` varchar(64) NOT NULL COMMENT '故障类型（如：灯具、水管、门窗）',
  `description` text NOT NULL COMMENT '详细故障描述',
  `images` varchar(1024) DEFAULT NULL COMMENT '图片链接（多张用逗号分隔，存OSS地址）',
  `address_snapshot` varchar(128) NOT NULL COMMENT '提交时的楼栋+宿舍号（如：8号楼-302）',
  `status` int NOT NULL DEFAULT '0' COMMENT '0:待处理, 1:已接单, 2:维修中, 3:已完成, 4:已取消',
  `worker_id` bigint DEFAULT NULL COMMENT '分派的维修员ID',
  `evaluation_score` int DEFAULT NULL COMMENT '评价分数（1-5）',
  `evaluation_content` varchar(255) DEFAULT NULL COMMENT '评价内容',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '报修提交时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '状态更新时间',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='报修订单表';

-- ----------------------------
-- 3. 维修员表
-- ----------------------------
CREATE TABLE `worker` (
  `id` bigint NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
  `username` varchar(32) NOT NULL COMMENT '维修员工号/登录账号',
  `password` varchar(128) NOT NULL COMMENT '密码',
  `name` varchar(32) NOT NULL COMMENT '真实姓名',
  `phone` varchar(11) DEFAULT NULL COMMENT '手机号',
  `skills` varchar(255) DEFAULT NULL COMMENT '擅长领域（如：电工、木工）',
  `is_available` int NOT NULL DEFAULT '1' COMMENT '是否在岗（1:是, 0:否）',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '修改时间',
  PRIMARY KEY (`id`),
  UNIQUE KEY `idx_worker_username` (`username`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='维修员表';

-- ----------------------------
-- 5. 设备资源表（洗衣机等）
-- ----------------------------
CREATE TABLE `device_resource` (

  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='设备资源表';

-- ----------------------------
-- 6. 生活服务预约表
-- ----------------------------

-- ----------------------------
-- 7. 失物招领表
-- ----------------------------
CREATE TABLE `lost_found` (
  `id` bigint NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
  `student_id` bigint NOT NULL COMMENT '发布者学生ID',
  `type` int NOT NULL COMMENT '信息类型（0: 寻物启事-我丢了东西, 1: 失物招领-我捡了东西）',
  `title` varchar(64) NOT NULL COMMENT '标题',
  `description` text NOT NULL COMMENT '物品详情描述',
  `category` varchar(32) DEFAULT NULL COMMENT '物品分类（如：校园卡、数码产品、书籍资料）',
  `location` varchar(128) DEFAULT NULL COMMENT '丢失或捡到的地点',
  `contact_info` varchar(64) NOT NULL COMMENT '联系方式（手机号或微信号）',
  `images` varchar(1024) DEFAULT NULL COMMENT '图片链接（逗号分隔，存OSS地址）',
  `status` int NOT NULL DEFAULT '0' COMMENT '状态（0: 寻找中/待认领, 1: 已解决, 2: 已撤销）',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '发布时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '最后修改时间',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='失物招领信息表';

-- ----------------------------
-- 8. 二手交易商品表
-- ----------------------------
CREATE TABLE `secondhand_item` (
  `id` bigint NOT NULL AUTO_INCREMENT COMMENT '主键，自增',
  `student_id` bigint NOT NULL COMMENT '卖家学生ID',
  `name` varchar(64) NOT NULL COMMENT '商品名称',
  `description` text NOT NULL COMMENT '商品详细描述',
  `category` varchar(32) DEFAULT NULL COMMENT '商品分类（如：电子数码、生活用品、代步工具）',
  `price` decimal(10,2) NOT NULL COMMENT '出售价格（精确到小数点后两位）',
  `condition_level` varchar(32) DEFAULT NULL COMMENT '成色（如：全新、9成新、有轻微伊拉克战损等）',
  `images` varchar(1024) DEFAULT NULL COMMENT '商品图片（多张逗号分隔）',
  `status` int NOT NULL DEFAULT '0' COMMENT '状态（0: 在售, 1: 已售出, 2: 已下架）',
  `create_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '发布时间',
  `update_time` datetime NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '修改时间',
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='二手交易商品表';
